#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
===============================================================================
ئەمەلىي قوللىنىشچان ئاپتوبۇس يولۇچى ساناش ۋە ئىز قوغلاش سىستېمىسى (YOLOv8 + OpenCV + SQLite)
===============================================================================
قاچىلاش بۇيرۇقى:
    pip install ultralytics opencv-python

ئىجرا قىلىش ئۇسۇلى:
    1) سىن ھۆججىتى بىلەن:  python bus_counter_pro.py --source "bus_door_cam.mp4"
    2) نەق مەيدان كامېرا:  python bus_counter_pro.py --source 0
===============================================================================
"""

# ==============================================================================
# 1-بۇيرۇق: مۇھىت تەڭشەش ۋە ئاساس كۈتۈپخانىلارنى سەپلەش
# ==============================================================================
import argparse
import sqlite3
from datetime import datetime
import cv2
from ultralytics import YOLO


# ==============================================================================
# 5-بۇيرۇق (ساندان قىسمى): SQLite ساندان باشقۇرغۇچىسى
# ==============================================================================
class PassengerDatabase:
    def __init__(self, db_path: str, ticket_fare: float):
        self.db_path = db_path
        self.ticket_fare = ticket_fare
        self.conn = sqlite3.connect(self.db_path, check_same_thread=False)
        self.cursor = self.conn.cursor()
        self._create_table()

    def _create_table(self) -> None:
        self.cursor.execute("""
            CREATE TABLE IF NOT EXISTS passenger_logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                track_id INTEGER NOT NULL,
                direction TEXT NOT NULL,
                line_sequence TEXT NOT NULL,
                x_center INTEGER NOT NULL,
                y_center INTEGER NOT NULL,
                fare REAL NOT NULL,
                timestamp TEXT NOT NULL
            )
        """)
        self.conn.commit()

    def insert_event(self, track_id: int, direction: str, x_c: int, y_c: int) -> None:
        now_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        seq = "A -> B" if direction == "IN" else "B -> A"
        fare_val = self.ticket_fare if direction == "IN" else 0.0
        self.cursor.execute(
            """
            INSERT INTO passenger_logs
            (track_id, direction, line_sequence, x_center, y_center, fare, timestamp)
            VALUES (?, ?, ?, ?, ?, ?, ?)
            """,
            (int(track_id), direction, seq, int(x_c), int(y_c), fare_val, now_str),
        )
        self.conn.commit()
        print(f"[{now_str}] SANAQ: ID=#{track_id} | {direction} ({seq}) | Kordinat=({x_c},{y_c}) | Kirim={fare_val:.2f}")

    def print_summary(self) -> None:
        self.cursor.execute("SELECT COUNT(*), COALESCE(SUM(fare), 0) FROM passenger_logs WHERE direction='IN'")
        total_in, total_rev = self.cursor.fetchone()
        self.cursor.execute("SELECT COUNT(*) FROM passenger_logs WHERE direction='OUT'")
        (total_out,) = self.cursor.fetchone()
        print("\n" + "=" * 60)
        print("SQLite ساندان جەمئىي ستاتىستىكا دوكلاتى:")
        print(f"  - جەمئىي كىرگەنلەر (IN) : {total_in} ئادەم")
        print(f"  - جەمئىي چىققانلار (OUT): {total_out} ئادەم")
        print(f"  - جەمئىي بېلەت كىرىمى  : {total_rev:.2f} سوم")
        print("=" * 60)

    def close(self) -> None:
        self.conn.close()


# ==============================================================================
# ئاساسىي قوللىنىشچان پروگرامما (2-، 3-، 4- ۋە 5-بۇيرۇقلار)
# ==============================================================================
def run_bus_counter(
    source: str = "bus_door_cam.mp4",
    model_weights: str = "yolov8n.pt",
    db_file: str = "bus_passengers.db",
    y_a: int = 145,
    y_b: int = 295,
    offset: int = 18,
    conf_thresh: float = 0.45,
    ticket_fare: float = 2.00,
    max_capacity: int = 5,
    save_output: str = "bus_counted_output.mp4",
) -> None:
    # 2-بۇيرۇق: YOLO مودېلىنى يۈكلەش ۋە سىن مەنبەسىنى تۇتاشتۇرۇش
    print(f"[1/3] YOLOv8 مودېلى يۈكلىنىۋاتىدۇ: {model_weights} ...")
    model = YOLO(model_weights)

    video_src = 0 if str(source).strip() == "0" else source
    cap = cv2.VideoCapture(video_src)
    if not cap.isOpened():
        raise RuntimeError(f"خاتالىق: سىن مەنبەسىنى ئوقۇغىلى بولمىدى -> {source}")

    frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    fps = int(cap.get(cv2.CAP_PROP_FPS)) or 30
    print(f"[2/3] سىن ئۇلاندى! ئۆلچىمى: {frame_width}x{frame_height} @ {fps} FPS")

    # ئەگەر بېكىتىلگەن سىزىق سىن ئېگىزلىكىدىن ئېشىپ كەتسە، نىسبەت بويىچە ئاپتوماتىك تەڭشەش
    if y_b >= frame_height or y_a >= frame_height:
        y_a = int(frame_height * 0.35)
        y_b = int(frame_height * 0.68)
        print(f"[ئەسكەرتىش] قوش سىزىق سىن ئېگىزلىكىگە ماسلاشتۇرۇلدى: Y_A={y_a}, Y_B={y_b}")

    # نەتىجە سىننى ساقلاش ئۈسكۈنىسى (VideoWriter)
    writer = None
    if save_output:
        fourcc = cv2.VideoWriter_fourcc(*"mp4v")
        writer = cv2.VideoWriter(save_output, fourcc, fps, (frame_width, frame_height))

    # SQLite سانداننى قوزغىتىش
    db = PassengerDatabase(db_file, ticket_fare)

    # 4-بۇيرۇق: قوش سىزىقلىق ھېسابلاش لۇغىتى ۋە سانالغانلار تىزىملىكى
    first_line_dict: dict[int, str] = {}   # {track_id: 'A' ياكى 'B'}
    counted_ids: set[int] = set()          # تەكرار ساناشنىڭ ئالدىنى ئېلىش توپلىمى
    in_count: int = 0
    out_count: int = 0

    print("[3/3] يولۇچى ساناش باشلاندى (چېكىنىش ئۈچۈن 'q' كۇنۇپكىسىنى بېسىڭ)...")

    while cap.isOpened():
        success, frame = cap.read()
        if not success:
            print("سىن ئاخىرلاشتى.")
            break

        # 3-بۇيرۇق: ئىز قوغلاش (Tracking) - پەقەت ئادەم (classes=[0])
        results = model.track(
            frame,
            persist=True,
            classes=[0],
            conf=conf_thresh,
            tracker="bytetrack.yaml",
            verbose=False,
        )

        # A سىزىق (تاشقى چېگرا) ۋە سەزگۈرلۈك رايونىنى سىزىش
        cv2.line(frame, (0, y_a), (frame_width, y_a), (0, 180, 255), 2)
        cv2.putText(
            frame,
            f"Line A (Outer) Y={y_a}",
            (16, max(20, y_a - 10)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.55,
            (0, 180, 255),
            2,
        )

        # B سىزىق (ئىچكى چېگرا) ۋە سەزگۈرلۈك رايونىنى سىزىش
        cv2.line(frame, (0, y_b), (frame_width, y_b), (120, 220, 0), 2)
        cv2.putText(
            frame,
            f"Line B (Inner) Y={y_b}",
            (16, max(20, y_b - 10)),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.55,
            (120, 220, 0),
            2,
        )

        # بايقالغان ئادەملەرنى بىر تەرەپ قىلىش
        if results[0].boxes is not None and results[0].boxes.id is not None:
            boxes = results[0].boxes.xyxy.cpu().numpy().astype(int)
            track_ids = results[0].boxes.id.cpu().numpy().astype(int)

            for box, track_id in zip(boxes, track_ids):
                x1, y1, x2, y2 = box
                x_center = int((x1 + x2) / 2)
                y_center = int((y1 + y2) / 2)

                # 4-بۇيرۇق: قوش سىزىقلىق ھېسابلاش لوگىكىسى
                if track_id not in counted_ids:
                    # A سىزىق سەزگۈرلۈك دائىرىسىگە كىردىمۇ؟
                    if (y_a - offset) <= y_center <= (y_a + offset):
                        if track_id not in first_line_dict:
                            first_line_dict[track_id] = "A"
                        elif first_line_dict[track_id] == "B":
                            # B -> A تەرتىپى تاماملاندى: چىقتى (OUT)
                            out_count += 1
                            counted_ids.add(track_id)
                            db.insert_event(track_id, "OUT", x_center, y_center)

                    # B سىزىق سەزگۈرلۈك دائىرىسىگە كىردىمۇ؟
                    elif (y_b - offset) <= y_center <= (y_b + offset):
                        if track_id not in first_line_dict:
                            first_line_dict[track_id] = "B"
                        elif first_line_dict[track_id] == "A":
                            # A -> B تەرتىپى تاماملاندى: كىردى (IN)
                            in_count += 1
                            counted_ids.add(track_id)
                            db.insert_event(track_id, "IN", x_center, y_center)

                # ئادەم رامكىسى، مەركىزىي نۇقتىسى ۋە track_id بەلگىسىنى سىزىش
                is_counted = track_id in counted_ids
                color = (80, 220, 100) if is_counted else (255, 190, 40)
                cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
                cv2.circle(frame, (x_center, y_center), 5, (0, 0, 255), -1)

                first_step = first_line_dict.get(track_id, "-")
                status_str = "COUNTED" if is_counted else f"First:{first_step}"
                cv2.putText(
                    frame,
                    f"ID:{track_id} [{status_str}]",
                    (x1, max(22, y1 - 8)),
                    cv2.FONT_HERSHEY_SIMPLEX,
                    0.52,
                    color,
                    2,
                )

        # 5-بۇيرۇق: ئېكرانغا ئادەم سانى، سىغىمچانلىق ئاگاھلاندۇرۇشى ۋە كىرىم مۆلچەرىنى چىقىرىش
        inside_bus = max(0, in_count - out_count)
        estimated_revenue = in_count * ticket_fare

        # يېرىم سۈزۈك ستاتىستىكا تاختىسى
        overlay = frame.copy()
        cv2.rectangle(overlay, (14, 14), (370, 180), (15, 23, 42), -1)
        cv2.addWeighted(overlay, 0.82, frame, 0.18, 0, frame)

        cv2.putText(frame, f"IN  (A->B) : {in_count}", (28, 46),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.72, (52, 211, 153), 2)
        cv2.putText(frame, f"OUT (B->A) : {out_count}", (28, 78),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.72, (56, 160, 255), 2)
        cv2.putText(frame, f"INSIDE BUS : {inside_bus}/{max_capacity}", (28, 110),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.72, (255, 210, 80), 2)
        cv2.putText(frame, f"REVENUE    : {estimated_revenue:.2f}", (28, 142),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.72, (80, 230, 255), 2)

        if inside_bus >= 0.9 * max_capacity:
            cv2.putText(frame, "ALERT: 90% CAPACITY REACHED!", (28, 170),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.65, (0, 0, 255), 2)

        if writer is not None:
            writer.write(frame)

        cv2.imshow("AI Bus Passenger Counter (YOLOv8 + SQLite)", frame)
        if cv2.waitKey(1) & 0xFF == ord("q"):
            break

    cap.release()
    if writer is not None:
        writer.release()
    cv2.destroyAllWindows()
    db.print_summary()
    db.close()


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="YOLOv8 Bus Passenger Counter")
    parser.add_argument("--source", type=str, default="bus_door_cam.mp4", help="Video path or camera index (0)")
    parser.add_argument("--model", type=str, default="yolov8n.pt", help="YOLOv8 model weights")
    parser.add_argument("--db", type=str, default="bus_passengers.db", help="SQLite database file")
    parser.add_argument("--ya", type=int, default=145, help="Outer Line A Y-coordinate")
    parser.add_argument("--yb", type=int, default=295, help="Inner Line B Y-coordinate")
    parser.add_argument("--offset", type=int, default=18, help="Line crossing sensitivity offset")
    parser.add_argument("--fare", type=float, default=2.00, help="Ticket price per passenger")
    args = parser.parse_args()

    run_bus_counter(
        source=args.source,
        model_weights=args.model,
        db_file=args.db,
        y_a=args.ya,
        y_b=args.yb,
        offset=args.offset,
        ticket_fare=args.fare,
    )
